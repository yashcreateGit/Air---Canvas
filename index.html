  <script>

    const video = document.getElementById('input_video');

    const canvas = document.getElementById('output_canvas');

    const ctx = canvas.getContext('2d');

    const paintCanvas = document.createElement('canvas');

    paintCanvas.width = 960;

    paintCanvas.height = 540;

    const pCtx = paintCanvas.getContext('2d');


    let currentColor = '#ff0044';

    let prevX = 0, prevY = 0;


    document.querySelectorAll('.color-btn').forEach(btn => {

      btn.addEventListener('click', () => {

        if(btn.dataset.color === 'CLEAR') {

          pCtx.clearRect(0, 0, paintCanvas.width, paintCanvas.height);

        } else {

          document.querySelectorAll('.color-btn').forEach(b => b.classList.remove('active'));

          btn.classList.add('active');

          currentColor = btn.dataset.color;

        }

      });

    });


    function onResults(results) {

      ctx.save();

      ctx.clearRect(0, 0, canvas.width, canvas.height);

      ctx.drawImage(paintCanvas, 0, 0, canvas.width, canvas.height);


      if (results.multiHandLandmarks && results.multiHandLandmarks.length > 0) {

        const lm = results.multiHandLandmarks[0];

        const ix = lm[8].x * canvas.width;

        const iy = lm[8].y * canvas.height;

        const mx = lm[12].x * canvas.width;

        const my = lm[12].y * canvas.height;


        const indexUp = lm[8].y < lm[6].y;

        const middleUp = lm[12].y < lm[10].y;

        const ringUp = lm[16].y < lm[14].y;

        const pinkyUp = lm[20].y < lm[18].y;


        // 1. Eraser Mode (4 fingers up)

        if (indexUp && middleUp && ringUp && pinkyUp) {

          pCtx.clearRect(ix - 30, iy - 30, 60, 60);

          ctx.strokeStyle = '#fff';

          ctx.strokeRect(ix - 30, iy - 30, 60, 60);

          prevX = 0; prevY = 0;

        }

        // 2. Hover / Select Mode (Index + Middle)

        else if (indexUp && middleUp && !ringUp && !pinkyUp) {

          prevX = 0; prevY = 0;

          ctx.beginPath();

          ctx.arc(ix, iy, 10, 0, 2 * Math.PI);

          ctx.fillStyle = currentColor;

          ctx.fill();


          if (iy < 60) {

            const el = document.elementFromPoint(canvas.getBoundingClientRect().right - (ix * (canvas.getBoundingClientRect().width / 960)), canvas.getBoundingClientRect().top + iy);

            if (el && el.classList.contains('color-btn')) el.click();

          }

        }

        // 3. Draw Mode (Index only)

        else if (indexUp && !middleUp && !ringUp && !pinkyUp) {

          ctx.beginPath();

          ctx.arc(ix, iy, 6, 0, 2 * Math.PI);

          ctx.fillStyle = currentColor;

          ctx.fill();


          if (iy > 60) {

            if (prevX !== 0 && prevY !== 0) {

              pCtx.strokeStyle = currentColor;

              pCtx.lineWidth = 6;

              pCtx.lineCap = 'round';

              pCtx.beginPath();

              pCtx.moveTo(prevX, prevY);

              pCtx.lineTo(ix, iy);

              pCtx.stroke();

            }

            prevX = ix; prevY = iy;

          }

        } else {

          prevX = 0; prevY = 0;

        }

      } else {

        prevX = 0; prevY = 0;

      }

      ctx.restore();

    }


    const hands = new Hands({locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`});

    hands.setOptions({maxNumHands: 1, modelComplexity: 1, minDetectionConfidence: 0.7, minTrackingConfidence: 0.7});

    hands.onResults(onResults);


    const camera = new Camera(video, {

      onFrame: async () => { await hands.send({image: video}); },

      width: 960,

      height: 540

    });

    camera.start();

  </script>
